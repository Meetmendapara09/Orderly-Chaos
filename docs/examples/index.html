1. Basic Trading Session

A complete example showing how to set up a book, place orders, and query state.

from orderly_chaos import LimitOrderBook

# Create order book
book = LimitOrderBook()

# Place limit orders
book.limit_sell(order_id=1, quantity=100, price=10250)
book.limit_buy(order_id=2, quantity=50, price=10240)

# Market orders
book.market_buy(order_id=3, quantity=75)

# Query state
print(f"Best Ask: ${book.best_sell() / 100:.2f}")
print(f"Total Volume: {book.volume()}")
print(f"Order Count: {book.count()}")

# Cancel an order
book.cancel(1)

2. Order Book Inspection

Inspect the state of the order book in detail.

from orderly_chaos import LimitOrderBook

book = LimitOrderBook()

# Add orders across multiple price levels
for i, (qty, price) in enumerate([
    (100, 100), (200, 101), (150, 102),  # Sell side
    (100, 99),  (200, 98),  (300, 97),  # Buy side
]):
    if i < 3:
        book.limit_sell(i + 1, qty, price)
    else:
        book.limit_buy(i + 1, qty, price)

# Inspect book levels
print("=== Order Book Summary ===")
print(f"Best Ask: {book.best_sell()}")
print(f"Best Bid: {book.best_buy()}")
print(f"Total Sell Volume: {book.volume_sell()}")
print(f"Total Buy Volume: {book.volume_buy()}")
print(f"Orders at price 101: {book.count_at(101)}")

3. Market Making Strategy

Simulate a basic market making strategy posting on both sides.

from orderly_chaos import LimitOrderBook
import random

book = LimitOrderBook()
next_order_id = 1

def post_market_making_orders(mid_price, spread=5, size=100):
    """Post buy and sell orders around a mid-price."""
    global next_order_id
    buy_price = mid_price - spread
    sell_price = mid_price + spread
    book.limit_buy(next_order_id, size, buy_price)
    next_order_id += 1
    book.limit_sell(next_order_id, size, sell_price)
    next_order_id += 1

def simulate_price_movement(book, steps=10):
    """Simulate incoming market orders causing price movements."""
    global next_order_id
    for _ in range(steps):
        side = random.choice([True, False])
        qty = random.randint(10, 50)
        if side:
            book.market_buy(next_order_id, qty)
        else:
            book.market_sell(next_order_id, qty)
        next_order_id += 1

# Initialize with market making orders
post_market_making_orders(mid_price=10250, spread=10, size=100)
print(f"Initial mid-price: {book.price()}")

# Simulate price movement
simulate_price_movement(book, steps=5)
print(f"After simulation: {book.price()}")

4. Matching Engine Simulation

Demonstrate how large orders cascade through the book.

from orderly_chaos import LimitOrderBook

book = LimitOrderBook()

# Set up initial liquidity
book.limit_sell(1, 100, 10250)
book.limit_buy(2, 50, 10240)

print("=== Matching Engine Simulation ===")

# Place a large buy that sweeps the ask
book.limit_buy(3, 200, 10260)
# This matches 100 @ 10250, leaving 100 resting at 10260

print(f"After large buy: {book.count()} orders")
print(f"  Best Ask: {book.best_sell()}")
print(f"  Buy vol at 10260: {book.volume_buy(10260)}")

# Partial fill
book.market_sell(4, 30)
# Matches 30 @ 10260

print(f"After market sell: {book.count()} orders")
print(f"  Remaining buy vol: {book.volume_buy(10260)}")
