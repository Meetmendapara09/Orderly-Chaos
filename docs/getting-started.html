Installation

Python (from PyPI)

pip install orderly-chaos

Python (from source)

git clone https://github.com/Meetmendapara09/Orderly-Chaos.git
cd Orderly-Chaos
bazel build //orderly_chaos:copy_dll
cp bazel-bin/orderly_chaos/lib_orderly_chaos.dll orderly_chaos/
pip install -e .

Build Requirements

The following dependencies are required:

  • C++ Compiler: GCC 7+ or Clang with C++17 support
  • Python: 3.5+ (for Python bindings)
  • Bazel: 7+ build system with Bzlmod (install guide)
  • On Windows: MinGW-w64 (custom toolchain in toolchain/) or MSVC

Quick Start

Python

from orderly_chaos import LimitOrderBook

book = LimitOrderBook()

# Place limit orders
book.limit_sell(order_id=1, quantity=100, price=10250)
book.limit_buy(order_id=2, quantity=50, price=10240)

# Market orders
book.market_buy(order_id=3, quantity=75)

# Query state
print(f"Best Ask: ${book.best_sell() / 100:.2f}")
print(f"Total Volume: {book.volume()}")

C++

#include "limit_order_book.hpp"
#include <iostream>

int main() {
    LOB::LimitOrderBook book;

    book.limit_sell(1, 100, 10250);
    book.limit_buy(2, 50, 10240);

    std::cout << "Volume: " << book.volume();
    return 0;
}

Python Guide

Creating a Limit Order Book

from orderly_chaos import LimitOrderBook
book = LimitOrderBook()

Placing Orders

# Limit orders
book.limit_sell(order_id=1, quantity=100, price=10250)
book.limit_buy(order_id=2, quantity=50, price=10240)
book.limit(side=True, order_id=3, quantity=75, price=10230)

# Market orders
book.market_buy(order_id=4, quantity=25)
book.market_sell(order_id=5, quantity=25)

Querying the Book

book.best_sell()        # best ask price
book.best_buy()         # best bid price
book.best(True)              # best price for side
book.volume()           # total volume
book.volume(10250)           # volume at price
book.count()            # order count
book.has(1)             # check order exists
book.cancel(1)          # cancel order
book.clear()            # clear all orders

C++ Guide

#include "limit_order_book.hpp"
#include <iostream>

using namespace LOB;

int main() {
    LimitOrderBook book;

    // Place limit orders
    book.limit_sell(1, 100, 10250);
    book.limit_buy(2, 50, 10240);

    // Market orders
    book.market_buy(3, 25);

    // Query
    std::cout << book.best_sell() << std::endl;
    std::cout << book.volume() << std::endl;

    // Cancel
    book.cancel(1);
    // Clear
    book.clear();

    return 0;
}

C API Guide

#include "lib_lob.hpp"

int main() {
    // Create book
    LOB::LimitOrderBook* book = new_();

    // Place orders
    limit_sell(book, 1, 100, 10250);
    limit_buy(book, 2, 50, 10240);

    // Market orders
    market_buy(book, 3, 25);

    // Query
    LOB::Price best_ask = best_sell(book);
    LOB::Volume vol = volume(book);

    // Cancel and cleanup
    cancel(book, 1);
    delete_(book);

    return 0;
}

make deployment
