Lifecycle Functions

LOB::LimitOrderBook* new_()
Create and return a pointer to a new, empty limit order book.
auto* book = new_(); // create
assert(count(book) == 0);
delete_(book);
void delete_(LOB::LimitOrderBook* book)
Delete the limit order book, freeing all associated memory.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
delete_(book); // frees all orders
ParameterTypeDescription
bookLimitOrderBook*Pointer to the book to delete
void clear(LOB::LimitOrderBook* book)
Clear all orders from the book.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_buy(book, 2, 50, 10240);
clear(book);
assert(count(book) == 0);
delete_(book);

Limit Order Functions

void limit_sell(LOB::LimitOrderBook* book, UID uid, Quantity quantity, Price price)
Place a sell limit order at the given price.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
assert(best_sell(book) == 10250);
delete_(book);
void limit_buy(LOB::LimitOrderBook* book, UID uid, Quantity quantity, Price price)
Place a buy limit order at the given price.
auto* book = new_();
limit_buy(book, 2, 50, 10240);
assert(best_buy(book) == 10240);
delete_(book);
void limit(LOB::LimitOrderBook* book, Side side, UID uid, Quantity quantity, Price price)
Place a limit order on the specified side.
auto* book = new_();
limit(book, true, 3, 75, 10240); // buy
limit(book, false, 4, 100, 10250); // sell
assert(count(book) == 2);
delete_(book);

Market Order Functions

void market_sell(LOB::LimitOrderBook* book, UID uid, Quantity quantity)
Execute a market sell against buy-side liquidity.
auto* book = new_();
limit_buy(book, 1, 100, 10240);
market_sell(book, 2, 40);
assert(volume_buy(book) == 60);
delete_(book);
void market_buy(LOB::LimitOrderBook* book, UID uid, Quantity quantity)
Execute a market buy against sell-side liquidity.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
market_buy(book, 2, 40);
assert(volume_sell(book) == 60);
delete_(book);
void market(LOB::LimitOrderBook* book, Side side, UID uid, Quantity quantity)
Execute a market order on the specified side.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_buy(book, 2, 100, 10240);
market(book, true, 3, 25);
assert(volume(book) == 150);
delete_(book);

Cancellation Functions

bool has(LOB::LimitOrderBook* book, UID uid)
Return true if the book contains an order with the given ID.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
assert(has(book, 1) == true);
assert(has(book, 99) == false);
delete_(book);
const LOB::Order* get(LOB::LimitOrderBook* book, UID uid)
Get a pointer to the order with the given ID.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
const LOB::Order* o = get(book, 1);
assert(o->price == 10250);
delete_(book);
void cancel(LOB::LimitOrderBook* book, UID uid)
Cancel an existing order by its ID.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
cancel(book, 1);
assert(count(book) == 0);
delete_(book);

Query Functions

Price best_sell(LOB::LimitOrderBook* book)
Return the best (lowest) sell/ask price. O(1).
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_sell(book, 2, 50, 10260);
assert(best_sell(book) == 10250);
delete_(book);
Price best_buy(LOB::LimitOrderBook* book)
Return the best (highest) buy/bid price. O(1).
auto* book = new_();
limit_buy(book, 1, 100, 10240);
limit_buy(book, 2, 50, 10230);
assert(best_buy(book) == 10240);
delete_(book);
Price best(LOB::LimitOrderBook* book, Side side)
Return the best price for the given side.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_buy(book, 2, 100, 10240);
assert(best(book, false) == 10250);
assert(best(book, true) == 10240);
delete_(book);
Volume volume_sell_price(LOB::LimitOrderBook* book, Price price)
Return the volume at a specific sell-side price.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
assert(volume_sell_price(book, 10250) == 100);
delete_(book);
Volume volume_sell(LOB::LimitOrderBook* book)
Return the total sell-side volume.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_sell(book, 2, 50, 10260);
assert(volume_sell(book) == 150);
delete_(book);
Volume volume_buy_price(LOB::LimitOrderBook* book, Price price)
Return the volume at a specific buy-side price.
auto* book = new_();
limit_buy(book, 1, 80, 10240);
assert(volume_buy_price(book, 10240) == 80);
delete_(book);
Volume volume_buy(LOB::LimitOrderBook* book)
Return the total buy-side volume.
auto* book = new_();
limit_buy(book, 1, 80, 10240);
limit_buy(book, 2, 60, 10230);
assert(volume_buy(book) == 140);
delete_(book);
Volume volume_price(LOB::LimitOrderBook* book, Price price)
Return the volume at a specific price across both sides.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_buy(book, 2, 75, 10240);
assert(volume_price(book, 10250) == 100);
assert(volume(book) == 175);
delete_(book);
Volume volume(LOB::LimitOrderBook* book)
Return the total volume across both sides.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_buy(book, 2, 75, 10240);
assert(volume(book) == 175);
delete_(book);
Count count_at(LOB::LimitOrderBook* book, Price price)
Return the number of orders at a specific price.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_sell(book, 2, 50, 10250);
assert(count_at(book, 10250) == 2);
delete_(book);
Count count_sell(LOB::LimitOrderBook* book)
Return the order count on the sell side.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_sell(book, 2, 50, 10260);
assert(count_sell(book) == 2);
delete_(book);
Count count_buy(LOB::LimitOrderBook* book)
Return the order count on the buy side.
auto* book = new_();
limit_buy(book, 1, 50, 10240);
assert(count_buy(book) == 1);
delete_(book);
Count count(LOB::LimitOrderBook* book)
Return the total number of orders in the book.
auto* book = new_();
limit_sell(book, 1, 100, 10250);
limit_buy(book, 2, 50, 10240);
assert(count(book) == 2);
delete_(book);
