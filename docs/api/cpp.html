Limit Order Methods

void limit_sell(UID order_id, Quantity quantity, Price price)
Add a new sell limit order. Matches against buy orders if price crosses the spread.
#include "limit_order_book.hpp"
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
assert(book.best_sell() == 10250);
ParameterTypeDescription
order_idUIDUnique order identifier
quantityQuantityNumber of shares to sell
pricePriceLimit price
void limit_buy(UID order_id, Quantity quantity, Price price)
Add a new buy limit order. Matches against sell orders if price crosses the spread.
LOB::LimitOrderBook book;
book.limit_buy(2, 50, 10240);
assert(book.best_buy() == 10240);
void limit(Side side, UID order_id, Quantity quantity, Price price)
Add a limit order to the specified side.
LOB::LimitOrderBook book;
book.limit(LOB::Side::Buy, 3, 75, 10240); // buy
book.limit(LOB::Side::Sell, 4, 100, 10250); // sell
assert(book.count() == 2);

Market Order Methods

void market_sell(UID order_id, Quantity quantity)
Execute a market sell order against buy-side liquidity.
LOB::LimitOrderBook book;
book.limit_buy(1, 100, 10240);
book.market_sell(2, 40); // sells 40 into bid
assert(book.volume_buy() == 60);
void market_buy(UID order_id, Quantity quantity)
Execute a market buy order against sell-side liquidity.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.market_buy(2, 40); // buys 40 from ask
assert(book.volume_sell() == 60);
void market(Side side, UID order_id, Quantity quantity)
Execute a market order on the specified side.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.limit_buy(2, 100, 10240);
book.market(LOB::Side::Buy, 3, 25);
book.market(LOB::Side::Sell, 4, 25);
assert(book.volume() == 150);

Query Methods

bool has(UID order_id) const
Check if an order with the given ID exists. O(1) average time.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
assert(book.has(1) == true);
assert(book.has(99) == false);
const Order& get(UID order_id)
Get a reference to the order with the given ID.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
const LOB::Order& o = book.get(1);
assert(o.price == 10250);
Price best_sell() const
Return the best (lowest) sell/ask price. Returns 0 if empty. O(1).
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.limit_sell(2, 50, 10260);
assert(book.best_sell() == 10250);
Price best_buy() const
Return the best (highest) buy/bid price. Returns 0 if empty. O(1).
LOB::LimitOrderBook book;
book.limit_buy(1, 100, 10240);
book.limit_buy(2, 50, 10230);
assert(book.best_buy() == 10240);
Price best(Side side) const
Return the best price for the given side.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.limit_buy(2, 100, 10240);
assert(book.best(LOB::Side::Sell) == 10250);
assert(book.best(LOB::Side::Buy) == 10240);
Price price() const
Return the current midpoint price. O(1).
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.limit_buy(2, 75, 10240);
assert(book.price() == 10245); // (10250+10240)/2
Price last_best_sell() const
Return the last best sell price before the most recent change.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
auto prev = book.last_best_sell();
Price last_best_buy() const
Return the last best buy price before the most recent change.
LOB::LimitOrderBook book;
book.limit_buy(1, 100, 10240);
auto prev = book.last_best_buy();

Volume & Count

Volume volume() const
Total volume across both sides.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.limit_buy(2, 75, 10240);
assert(book.volume() == 175);
assert(book.volume(10250) == 100);
Volume volume_sell() const / volume_sell(Price price) const
Sell-side volume, total or at price.
assert(book.volume_sell() == 100);
assert(book.volume_sell(10250) == 100);
Volume volume_buy() const / volume_buy(Price price) const
Buy-side volume, total or at price.
assert(book.volume_buy() == 75);
assert(book.volume_buy(10240) == 75);
Count count() const
Total number of orders.
assert(book.count() == 2);
assert(book.count_sell() == 1);
assert(book.count_buy() == 1);
assert(book.count_at(10250) == 1);

Management Methods

void cancel(UID order_id)
Cancel an existing order. O(log n) due to tree removal.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.cancel(1);
assert(book.count() == 0);
void reduce(UID order_id, Quantity quantity)
Reduce the quantity of an existing order by the specified amount.
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.reduce(1, 40);
assert(book.volume_sell() == 60);
void clear()
Remove all orders from both sides of the book. O(n).
LOB::LimitOrderBook book;
book.limit_sell(1, 100, 10250);
book.limit_buy(2, 50, 10240);
book.clear();
assert(book.count() == 0);

Structures

LOB::Order

struct Order {
    UID uid;           // unique order ID
    Side side;         // buy or sell
    Quantity quantity; // shares in this order
    Price price;       // price of the order
    Limit* limit;      // pointer to containing price level
    Order* prev;       // previous order at this price level
    Order* next;       // next order at this price level
};

LOB::Limit

struct Limit {
    Price key;             // price level
    Volume volume;         // total volume at this price
    Count count;           // orders at this price
    Order* head;           // first order (time-priority queue)
    Order* tail;           // last order
    Limit* next;           // next price level
    Limit* prev;           // previous price level
};
